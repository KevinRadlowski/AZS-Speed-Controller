cmake_minimum_required(VERSION 3.21)
project(AZSSpeed VERSION 1.3.1 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Portable binaries: statically link MSVC C/C++ runtime in Release.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
FetchContent_Declare(minhook GIT_REPOSITORY https://github.com/TsudaKageyu/minhook.git GIT_TAG v1.3.4)
FetchContent_MakeAvailable(minhook)
add_library(AZSSpeedHook SHARED src/hook.cpp src/timer_fix.cpp)
set_property(TARGET AZSSpeedHook PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(AZSSpeedHook PRIVATE minhook winmm bcrypt)
set_target_properties(AZSSpeedHook PROPERTIES OUTPUT_NAME "AZSSpeedHook" PREFIX "")
add_executable(AZSSpeedController WIN32 src/controller.cpp src/app.rc)
set_property(TARGET AZSSpeedController PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(AZSSpeedController PRIVATE dwmapi comctl32)
target_include_directories(AZSSpeedController PRIVATE src)

if(MINGW)
  target_link_options(AZSSpeedHook PRIVATE -static -static-libgcc -static-libstdc++)
  target_link_options(AZSSpeedController PRIVATE -static -static-libgcc -static-libstdc++ -municode)
endif()

if(MSVC)
  target_compile_options(AZSSpeedController PRIVATE /utf-8)
  target_compile_options(AZSSpeedHook PRIVATE /utf-8)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(AZSTimerPolicyTest tests/timer_policy_test.cpp)
  add_test(NAME TimerPolicy COMMAND AZSTimerPolicyTest)
  if(MINGW)
    target_link_options(AZSTimerPolicyTest PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endif()
